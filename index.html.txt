<script>
const STORAGE_KEY="assignments-app";
const STATUS=["unconfirmed","confirmed","no_answer","need_replacement"];

const urgencyOrder={need_replacement:1,no_answer:2,confirmed:3,unconfirmed:4};
let sortByUrgency=false;

const sectionsData=[
{id:"opening",class:"opening",title:"Opening",items:["Chairman","Initial Prayer"]},
{id:"tesoros",class:"tesoros",title:"Tesoros de la Biblia",items:["Tesoros","Perlas Escondidas","Lectura"]},
{id:"maestros",class:"maestros",title:"Seamos Mejores Maestros",items:["1","2","3","4"]},
{id:"vida",class:"vida",title:"Nuestra Vida Cristiana",items:["1","2","3"]},
{id:"closing",class:"opening",title:"Closing",items:["Final Prayer"]}
];

let state=JSON.parse(localStorage.getItem(STORAGE_KEY))||{};
function save(){localStorage.setItem(STORAGE_KEY,JSON.stringify(state));}

function render(){
const root=document.getElementById("sections");
const overview=document.getElementById("overviewList");
root.innerHTML="";overview.innerHTML="";
let total=0,confirmed=0;
let overviewItems=[];

sectionsData.forEach(sec=>{
const secDiv=document.createElement("div");
secDiv.className="section "+sec.class;
secDiv.innerHTML=`<h2>${sec.title}</h2>`;

sec.items.forEach(name=>{
const key=sec.id+name;
if(!state[key]) state[key]={name:"",status:"unconfirmed",notes:""};
const item=state[key];

if(item.name){
overviewItems.push({name,person:item.name,status:item.status});
}

total++;
if(item.status==="confirmed") confirmed++;

const div=document.createElement("div");
div.className="assignment"+(item.status==="unconfirmed"?" unconfirmed-highlight":"");

div.innerHTML=`
<strong>${name}</strong>
<input value="${item.name}" placeholder="Name" />
<div class="status-row"></div>
<textarea>${item.notes}</textarea>
`;

const input=div.querySelector("input");
input.oninput=e=>{item.name=e.target.value;save();render();};

const notes=div.querySelector("textarea");
notes.oninput=e=>{item.notes=e.target.value;save();};

const row=div.querySelector(".status-row");
STATUS.forEach(s=>{
const btn=document.createElement("button");
btn.textContent=s.replace("_"," ");
btn.className=`status-btn status-${s} ${item.status===s?"":"inactive"}`;
btn.onclick=()=>{item.status=s;save();render();};
row.appendChild(btn);
});

secDiv.appendChild(div);
});
root.appendChild(secDiv);
});

if(sortByUrgency){
overviewItems.sort((a,b)=>urgencyOrder[a.status]-urgencyOrder[b.status]);
}

overviewItems.forEach(i=>{
const line=document.createElement("div");
line.innerHTML=`${i.name}: ${i.person} <span class="tag ${i.status}">${i.status.replace("_"," ")}</span>`;
overview.appendChild(line);
});

const percent=Math.round((confirmed/total)*100)||0;
document.getElementById("progressText").textContent=`${confirmed}/${total} confirmed (${percent}%)`;
document.getElementById("progressFill").style.width=percent+"%";
}

// Buttons

document.getElementById("copyBtn").onclick=()=>{
let text="";
document.querySelectorAll("#overviewList div").forEach(d=>{
text+=d.innerText+"\n";
});
navigator.clipboard.writeText(text);
alert("Overview copied!");
};

document.getElementById("sortBtn").onclick=()=>{
sortByUrgency=!sortByUrgency;
render();
};

document.getElementById("resetBtn").onclick=()=>{
if(confirm("Are you sure you want to reset everything?")){
state={};save();render();
}}

document.getElementById("clearBtn").onclick=()=>{
Object.values(state).forEach(i=>{
if(i.status==="confirmed"){i.name="";i.notes="";i.status="unconfirmed"}
});save();render();
}

render();
</script>
